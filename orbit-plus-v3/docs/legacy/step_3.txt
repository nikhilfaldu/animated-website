STEP 2: MEASURE CARD RECTANGLES

Create: src/sections/WorkBridge/useWorkCardRects.ts

Behavior:

	•	On activation, measure immediately, then on resize and scroll.
	•	Wait for fonts to be ready before final “ready” signal.

// src/sections/WorkBridge/useWorkCardRects.ts
import { useEffect, useMemo, useState } from 'react';

export type Rect = { x: number; y: number; w: number; h: number; r: number };

export function useWorkCardRects(projectIds: string[], active: boolean) {
  const [rects, setRects] = useState<Record<string, Rect>>({});
  const [ready, setReady] = useState(false);

  const ids = useMemo(() => projectIds.slice(0, 4), [projectIds]);

  useEffect(() => {
    if (!active) return;
    if (!ids.length) return;

    let raf = 0;

    function measure() {
      const next: Record<string, Rect> = {};

      for (const id of ids) {
        const el = document.getElementById(`work-card-${id}`);
        if (!el) continue;
        const b = el.getBoundingClientRect();

        next[id] = {
          x: b.left,
          y: b.top,
          w: b.width,
          h: b.height,
          r: 16,
        };
      }

      setRects(next);
      setReady(ids.every((id) => !!next[id]));
    }

    const measureSoon = () => {
      if (raf) cancelAnimationFrame(raf);
      raf = requestAnimationFrame(measure);
    };

    const run = async () => {
      if ('fonts' in document) {
        try {
          // @ts-expect-error fonts exists in modern browsers
          await document.fonts.ready;
        } catch {
          // ignore
        }
      }
      measureSoon();
    };

    run();

    const ro = new ResizeObserver(() => measureSoon());
    ro.observe(document.documentElement);

    window.addEventListener('resize', measureSoon);
    window.addEventListener('scroll', measureSoon, { passive: true });

    return () => {
      if (raf) cancelAnimationFrame(raf);
      ro.disconnect();
      window.removeEventListener('resize', measureSoon);
      window.removeEventListener('scroll', measureSoon);
    };
  }, [ids, active]);

  return { rects, ready };
}

STEP 5: BRIDGE PROGRESS HOOK (GLOBAL IN, MOTIONVALUES OUT)

Create: src/sections/WorkBridge/useBridgeProgress.ts

// src/sections/WorkBridge/useBridgeProgress.ts
import { MotionValue, useTransform } from 'framer-motion';
import {
  BRIDGE_GLOBAL_START,
  BRIDGE_GLOBAL_LOCK,
  BRIDGE_MORPH_START,
  BRIDGE_MORPH_END,
  BRIDGE_FADE_OUT_END,
} from './bridgeConstants';

function clamp01(n: number) {
  return Math.max(0, Math.min(1, n));
}

export function useBridgeProgress(globalProgress: MotionValue<number>) {
  const hint = useTransform(globalProgress, (v) => {
    const p = (v - BRIDGE_GLOBAL_START) / (BRIDGE_GLOBAL_LOCK - BRIDGE_GLOBAL_START);
    return clamp01(p);
  });

  const morph = useTransform(globalProgress, (v) => {
    const p = (v - BRIDGE_MORPH_START) / (BRIDGE_MORPH_END - BRIDGE_MORPH_START);
    return clamp01(p);
  });

  const opacity = useTransform(globalProgress, (v) => {
    if (v < BRIDGE_GLOBAL_START) return 0;
    if (v < BRIDGE_MORPH_END) return 1;
    const p = (v - BRIDGE_MORPH_END) / (BRIDGE_FADE_OUT_END - BRIDGE_MORPH_END);
    return 1 - clamp01(p);
  });

  return { hint, morph, opacity };
}

STEP 6: CREATE THE OVERLAY COMPONENT (FOUR REAL PREVIEWS, NOT ONE STACKED IMAGE)

Create: src/sections/WorkBridge/WorkBridgeOverlay.tsx

This implementation:
	•	Renders a dark plate
	•	Creates 4 animated rounded-rect clip paths
	•	Renders 4 separate  layers, each clipped by its own animated rect
	•	Keeps OrbitSystem emblem visible above (overlay z 55, emblem z 60)

// src/sections/WorkBridge/WorkBridgeOverlay.tsx
import { MotionValue, motion, useReducedMotion, useTransform } from 'framer-motion';
import { useMemo } from 'react';
import { useBridgeProgress } from './useBridgeProgress';
import type { Rect } from './useWorkCardRects';
import type { ProjectPreview } from './bridgeTypes';

type Props = {
  globalProgress: MotionValue<number>;
  projects: ProjectPreview[]; // first 4 in grid order
  rects: Record<string, Rect>;
  rectsReady: boolean;
};

function lerp(a: number, b: number, t: number) {
  return a + (b - a) * t;
}

export function WorkBridgeOverlay({ globalProgress, projects, rects, rectsReady }: Props) {
  const prefersReducedMotion = useReducedMotion();
  const { hint, morph, opacity } = useBridgeProgress(globalProgress);

  if (prefersReducedMotion) return null;
  if (!projects || projects.length < 4) return null;

  const startRects = useMemo(() => {
    const vw = window.innerWidth;
    const vh = window.innerHeight;

    const size = Math.min(vw, vh) < 640 ? 150 : 210;
    const gap = Math.min(vw, vh) < 640 ? 14 : 18;
    const r = 18;

    const cx = vw / 2;
    const cy = vh / 2;

    return [
      { x: cx - gap / 2 - size, y: cy - gap / 2 - size, w: size, h: size, r },
      { x: cx + gap / 2,        y: cy - gap / 2 - size, w: size, h: size, r },
      { x: cx - gap / 2 - size, y: cy + gap / 2,        w: size, h: size, r },
      { x: cx + gap / 2,        y: cy + gap / 2,        w: size, h: size, r },
    ];
  }, []);

  const windows = projects.slice(0, 4).map((p, i) => {
    const end = rects[p.id];
    const endRect = rectsReady && end ? end : startRects[i];

    const x = useTransform(morph, (t) => lerp(startRects[i].x, endRect.x, t));
    const y = useTransform(morph, (t) => lerp(startRects[i].y, endRect.y, t));
    const w = useTransform(morph, (t) => lerp(startRects[i].w, endRect.w, t));
    const h = useTransform(morph, (t) => lerp(startRects[i].h, endRect.h, t));
    const r = useTransform(morph, (t) => lerp(startRects[i].r, endRect.r ?? startRects[i].r, t));

    return { id: p.id, i, x, y, w, h, r, preview: p };
  });

  const blurPx = useTransform(hint, (v) => `${Math.max(0, 12 - v * 10)}px`);
  const windowAlpha = useTransform(hint, (v) => 0.2 + 0.8 * v);

  return (
    <motion.div
      aria-hidden="true"
      className="fixed inset-0 pointer-events-none"
      style={{ zIndex: 55, opacity }}
    >
      <svg className="absolute inset-0 h-full w-full" preserveAspectRatio="none">
        <defs>
          {windows.map((win) => (
            <clipPath key={win.id} id={`wb-clip-${win.id}`}>
              <motion.rect
                x={win.x}
                y={win.y}
                width={win.w}
                height={win.h}
                rx={win.r}
                ry={win.r}
              />
            </clipPath>
          ))}
        </defs>

        <rect x="0" y="0" width="100%" height="100%" fill="rgba(0,0,0,0.92)" />

        {windows.map((win) => (
          <g key={win.id} clipPath={`url(#wb-clip-${win.id})`}>
            <motion.image
              href={win.preview.previewImageUrl}
              x="0"
              y="0"
              width="100%"
              height="100%"
              preserveAspectRatio="xMidYMid slice"
              style={{ filter: blurPx, opacity: windowAlpha }}
            />
          </g>
        ))}

        <rect x="0" y="0" width="100%" height="100%" fill="rgba(255,255,255,0.02)" />
      </svg>
    </motion.div>
  );
}

import { useEffect, useMemo, useState } from 'react';
import { useMotionValueEvent } from 'framer-motion';
import { WorkBridgeOverlay, useWorkCardRects, BRIDGE_GLOBAL_START, BRIDGE_FADE_OUT_END } from '@/sections/WorkBridge';
import { PROJECTS } from '@/sections/Work';

import { useGlobalScrollProgress } from '@/components/ScrollController/useGlobalScrollProgress'; // must be your existing MotionValue

export function App() {
  const globalProgress = useGlobalScrollProgress();

  const bridgeProjects = useMemo(() => {
    return PROJECTS.slice(0, 4).map((p) => ({
      id: p.id,
      previewImageUrl: p.previewImageUrl,
      previewFocalPoint: p.previewFocalPoint,
    }));
  }, []);

  const [bridgeActive, setBridgeActive] = useState(false);

  useMotionValueEvent(globalProgress, 'change', (v) => {
    setBridgeActive(v >= BRIDGE_GLOBAL_START && v <= BRIDGE_FADE_OUT_END);
  });

  const { rects, ready } = useWorkCardRects(
    bridgeProjects.map((p) => p.id),
    bridgeActive
  );

  useEffect(() => {
    for (const p of PROJECTS.slice(0, 4)) {
      const img = new Image();
      img.decoding = 'async';
      img.src = p.previewImageUrl;
    }
  }, []);

  return (
    <>
      {/* … your fixed OrbitSystem emblem layer … */}

      <WorkBridgeOverlay
        globalProgress={globalProgress}
        projects={bridgeProjects}
        rects={rects}
        rectsReady={ready}
      />

      {/* … your sections including <Reveal /> and <Work /> … */}
    </>
  );
}
