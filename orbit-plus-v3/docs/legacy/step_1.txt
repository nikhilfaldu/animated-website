Animation Mechanics

Shutter Blade Physics

Each of the 32 rays pivots around its inner anchor (near emblem center):

Initial (progress=0):
	•	Rays rotated 90° outward from final position
	•	Positioned far offscreen (radius ~800px)

Animation (0→1):
	•	Spring interpolation: stiffness=180, damping=22
	•	Creates visible overshoot (~8°) then settle
	•	Quadrant-distributed stagger (12ms per ray)

Final (progress=1):
	•	All rays at origin, 0° rotation
	•	Matches OrbitSystem “idle” state exactly

Vertex Shader Hinge Rotation

// Translate to hinge point, rotate, translate back
vec3 pos = position;
pos.xy -= hingeAnchor;
pos.xy = rotate2D(pos.xy, swingAngle);
pos.xy += hingeAnchor;
pos.xy += radialOffset * (1.0 - progress); // Converge to center

Scroll Mapping

const APERTURE_START = 0.70;  // Animation begins
const APERTURE_END = 0.95;    // Animation complete
const CROSSFADE = 0.03;       // 3% scroll for handoff blend

// Maps section scroll → animation progress
function mapProgress(sectionScroll: number): number {
return clamp((sectionScroll - 0.70) / 0.25, 0, 1);
}

export type OrbitRect = { x: number; y: number; width: number; height: number } | null;

export function useOrbitEmblemRect() {

src/components/emblem/ApertureEmblem/apertureShader.ts

export const vertexShader = varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); };

export const fragmentShader = `
precision highp float;

uniform sampler2D uTex;
uniform float uProgress;   // 0..1
uniform float uOpacity;    // 0..1
uniform float uSwingMax;   // radians (1.57 = 90deg)
uniform float uRadialPush; // radial push at start
uniform float uHinge;      // hinge radius in normalized coords (0..1)
uniform float uDelays[32]; // per-slice delay

varying vec2 vUv;

const float TAU = 6.283185307179586;
const float SLICES = 32.0;

mat2 rot(float a) {
float s = sin(a);
float c = cos(a);
return mat2(c, -s, s, c);
}

float saturate(float x) { return clamp(x, 0.0, 1.0); }

void main() {
vec2 c = (vUv - 0.5) * 2.0;

float theta = atan(c.y, c.x);
if (theta < 0.0) theta += TAU;

float sliceW = TAU / SLICES;
float slice = floor(theta / sliceW);
float sliceCenter = (slice + 0.5) * sliceW;

int idx = int(slice);
float d = uDelays[idx];

float p = saturate((uProgress - d) / max(0.0001, (1.0 - d)));
p = smoothstep(0.0, 1.0, p);

vec2 cs = rot(-sliceCenter) * c;

vec2 hinge = vec2(uHinge, 0.0);

float swing = (1.0 - p) * uSwingMax;

cs.x += uRadialPush * (1.0 - p);

vec2 rel = cs - hinge;
rel = rot(swing) * rel;
cs = rel + hinge;

vec2 cw = rot(sliceCenter) * cs;

vec2 uv2 = (cw * 0.5) + 0.5;

if (uv2.x < 0.0 || uv2.x > 1.0 || uv2.y < 0.0 || uv2.y > 1.0) {
  gl_FragColor = vec4(0.0);
  return;
}

vec4 tex = texture2D(uTex, uv2);

float a = smoothstep(0.15, 0.35, tex.a);

gl_FragColor = vec4(vec3(1.0), a * uOpacity);

}
`;

function clamp01(n: number) {
return Math.max(0, Math.min(1, n));
}

export function useApertureUniforms(scrollProgress: any, invalidate: () => void) {
const uniforms = useMemo(() => {
const delays = new Float32Array(32);

// Quadrant distributed stagger with deterministic jitter (no Math.random)
const order: number[] = [];
const quadrants = [
  Array.from({ length: 8 }, (_, i) => i),
  Array.from({ length: 8 }, (_, i) => i + 8),
  Array.from({ length: 8 }, (_, i) => i + 16),
  Array.from({ length: 8 }, (_, i) => i + 24),
];

const stride = 3;
const shuffled = quadrants.map((q) => q.map((_, i) => q[(i * stride) % q.length]));

// Quadrant order: NE → SW → NW → SE
for (let i = 0; i < 8; i++) {
  order.push(shuffled[0][i], shuffled[2][i], shuffled[1][i], shuffled[3][i]);
}

order.forEach((rayIdx, i) => {
  const base = i * 0.010;              // 10ms base spacing
  const jitter = (rayIdx % 7) * 0.002; // 0..12ms deterministic jitter
  delays[rayIdx] = Math.min(0.22, base + jitter);
});

return {
  uTex: { value: null as unknown as THREE.Texture },
  uProgress: { value: 0 },
  uOpacity: { value: 0 },
  uSwingMax: { value: Math.PI / 2 },
  uRadialPush: { value: 0.55 },
  uHinge: { value: 0.08 },
  uDelays: { value: Array.from(delays) },
};

}, []);

const raw = useTransform(scrollProgress, (v: number) => {
const p = (v - APERTURE_START) / (APERTURE_END - APERTURE_START);
return clamp01(p);
});

const spring = useSpring(raw, { stiffness: 180, damping: 22, mass: 1 });

const opacityMv = useTransform(scrollProgress, (v: number) => {
if (v < APERTURE_START - 0.02) return 0;
if (v < APERTURE_START) return (v - (APERTURE_START - 0.02)) / 0.02;
if (v < WEBGL_FADE_OUT_START) return 1;
if (v < WEBGL_FADE_OUT_END) {
return 1 - (v - WEBGL_FADE_OUT_START) / (WEBGL_FADE_OUT_END - WEBGL_FADE_OUT_START);
}
return 0;
});

const textureLoadedRef = useRef(false);
useEffect(() => {
if (textureLoadedRef.current) return;
textureLoadedRef.current = true;

(async () => {
  const tex = await createEmblemTexture(2048);
  uniforms.uTex.value = tex;
  invalidate();
})().catch((err) => console.error(err));

}, [uniforms, invalidate]);

useMotionValueEvent(spring, ‘change’, (v) => {
uniforms.uProgress.value = v;
invalidate();
});

useMotionValueEvent(opacityMv, ‘change’, (v) => {
uniforms.uOpacity.value = v;
invalidate();
});

return uniforms;
}
