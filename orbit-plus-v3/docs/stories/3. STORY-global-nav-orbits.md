# docs/stories/STORY-global-nav-orbits.md

## Title

Global Navigation — Orbital Resonance (Emblem ⇄ Dial Morph)

## Goal

Unify the emblem (with scroll progress) and the dial nav into a **single object with two modes**: Compact (emblem) and Pinned (dial). The emblem’s progress arc **becomes** the dial’s active selection arc, with a continuous morph and robust a11y. 60 FPS, GPU-only transforms, reduced-motion parity.

## Narrative Link

Act 2: Orbit — the emblem becomes a navigational compass; nodes represent available trajectories through the narrative.

## Files to Touch

* `src/components/navigation/GlobalNav.tsx` — Orchestrator (modes, ARIA, roving index, scroll handoff)
* `src/components/navigation/KineticDialNav.tsx` — Pure visual ring (angles → positions)
* `src/components/emblem/Emblem.tsx` — Share origin/scale/rotate motion values for morph (compact⇄dial)
* `src/context/OrbitSystem.tsx` — Expose `scrollProgress`/`scrollVelocity`/transforms (hand → arc sync)
* `src/sections/index.ts` — Ordered section registry (map to dial items)
* Respect V2 parity for “open on hover, pin on click” & placement from checklist
* Use Lenis for scrollTo timing & anchor parity

## State Machine (authoritative)

```
compact → preview (hover/focus emblem)
preview → compact (blur/leave unless pinned)
preview → pinned (click/tap emblem)
pinned  → navigating (item select → lenis.scrollTo)
navigating → pinned (until auto-collapse timer)
pinned  → compact (escape / click-away / auto-collapse after 600ms idle)
```

`mode: 'compact' | 'preview' | 'pinned'` lives in `GlobalNav`.

## Geometry & Mapping

**Angles**
`angleDeg = index * (360 / N)`; keep **document order** (Hero→…→Contact) for cognitive alignment.

**Progress ⇄ Angle**

* Read `scrollProgress` from OrbitSystem (0→1 across entire scroll).
* Snap to nearest section index using precomputed section offsets.
* While **pinned**, the progress hand becomes a **selection hand**; freeze it unless user selects another item.

---

## Transformation Choreography (compact → preview → pinned)

**Timings**

* Prime 80–120 ms → Unfurl 220–300 ms → Settle 140–200 ms
* Easing for Orbit feel: `cubic-bezier(0.33, 1, 0.68, 1)`
* Collapse uses ~⅔ durations to feel snappier.

**Shared Motion Values** (from Emblem)
Use PRD’s dynamic origin pattern during morph (center→nav origin). The emblem container itself remains `pointer-events-none`; a small internal hit area receives events (nav safety).

## Orchestrator with A11y & Roving Index

```tsx
// GlobalNav.tsx (orchestrator):contentReference[oaicite:17]{index=17}
import { useEffect, useMemo, useRef, useState } from 'react';
import { motion, useSpring, useTransform } from 'framer-motion';
import { useLenis } from '@/components/LenisProvider';                //:contentReference[oaicite:18]{index=18}
import { useOrbitSystem } from '@/context/OrbitSystem';               //:contentReference[oaicite:19]{index=19}
import sections from '@sections';                                     // ordered registry:contentReference[oaicite:20]{index=20}

type NavMode = 'compact' | 'preview' | 'pinned';

export function GlobalNav() {
  const { lenis } = useLenis();
  const { scrollProgress, originX, originY, scale } = useOrbitSystem(); // MVs:contentReference[oaicite:21]{index=21}
  const [mode, setMode] = useState<NavMode>('compact');
  const [activeIndex, setActiveIndex] = useState(0);
  const [expanded, setExpanded] = useState(false); // aria-expanded on hit area

  // 1) Build items & angles from sections (document order)
  const items = useMemo(() => {
    const N = sections.length;
    return sections.map((s, i) => ({
      id: `#${s.id}`, label: s.title, angleDeg: (360 / N) * i
    }));
  }, []);

  // 2) Map progress → nearest section index
  const sectionOffsets = useMemo(() => computeSectionOffsets(sections), []);
  useEffect(() => {
    const unsub = scrollProgress.on('change', (p: number) => {
      setActiveIndex(findNearestSectionIndex(p, sectionOffsets));
    });
    return () => unsub();
  }, [scrollProgress, sectionOffsets]);

  // 3) Morph controls (shared with Emblem): small prime on preview
  const previewScale = useSpring(1, { stiffness: 140, damping: 22 });
  useEffect(() => {
    if (mode === 'preview' || mode === 'pinned') previewScale.set(1.04);
    else previewScale.set(1.0);
  }, [mode]);

  // 4) Events (hover/focus → preview; click → pinned)
  const openPreview = () => setMode(m => (m === 'compact' ? 'preview' : m));
  const closePreview = () => setMode(m => (m === 'preview' ? 'compact' : m));
  const togglePinned = () => setMode(m => (m === 'pinned' ? 'compact' : 'pinned'));

  // 5) A11y: roving index (left/right). Escape collapses.
  const ringRef = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (mode === 'compact') return;
      if (e.key === 'Escape') { setMode('compact'); setExpanded(false); }
      if (e.key === 'ArrowRight') setActiveIndex(i => (i + 1) % items.length);
      if (e.key === 'ArrowLeft')  setActiveIndex(i => (i - 1 + items.length) % items.length);
      if (e.key === 'Enter' || e.key === ' ') handleSelect(items[activeIndex].id);
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [mode, items, activeIndex]);

  // 6) Select → scroll + pin behavior
  const handleSelect = (id: string) => {
    setMode('pinned');
    setExpanded(true);
    lenis?.scrollTo(id, { duration: 1.2 }); // easing matches Lenis default:contentReference[oaicite:22]{index=22}
    // Auto-collapse after idle
    window.setTimeout(() => { setMode('compact'); setExpanded(false); }, 600);
  };

  // 7) ARIA & hit area (keeps emblem pointer-safe)
  return (
    <div
      className="fixed left-0 top-32 z-50"                             // V2 parity position:contentReference[oaicite:23]{index=23}
      aria-label="Global navigation"
    >
      <button
        id="emblem-trigger"
        className="relative z-10 h-12 w-12 rounded-full"
        aria-expanded={expanded}
        aria-controls="global-nav"
        onMouseEnter={openPreview}
        onFocus={openPreview}
        onMouseLeave={closePreview}
        onClick={() => { togglePinned(); setExpanded(m => !m); }}
      >
        <motion.span
          className="sr-only"
          animate={{ opacity: mode === 'compact' ? 1 : 0.001 }}
        >Open navigation</motion.span>
        {/* Invisible hit area aligned visually with emblem; emblem itself remains pointer-events:none */}
      </button>

      <KineticDialNav
        ref={ringRef}
        id="global-nav"
        mode={mode}
        items={items}
        activeIndex={activeIndex}
        onHoverIndex={i => setActiveIndex(i)}
        onSelectIndex={i => handleSelect(items[i].id)}
        shared={{ originX, originY, scale: previewScale }}            // share morph controls
      />
    </div>
  );
}

// Helpers (compute once; keep simple to avoid runtime layout thrash)
function computeSectionOffsets(sections: { id: string }[]) {
  // Map each section to [start, end] in normalized scroll (0..1).
  // For now: equal segments as a placeholder; refine with actual DOM on mount.
  const N = sections.length;
  return sections.map((_, i) => ({ start: i / N, end: (i + 1) / N }));
}

function findNearestSectionIndex(p: number, ranges: { start: number; end: number }[]) {
  let best = 0, bestD = Infinity;
  ranges.forEach((r, i) => {
    const mid = (r.start + r.end) / 2;
    const d = Math.abs(p - mid);
    if (d < bestD) { bestD = d; best = i; }
  });
  return best;
}
```

## Pure Dial View (polar geometry, GPU-only)

```tsx
// KineticDialNav.tsx (pure view):contentReference[oaicite:24]{index=24}
import { forwardRef } from 'react';
import { motion, useTransform } from 'framer-motion';
import clsx from 'clsx';

type Props = {
  id: string;
  mode: 'compact' | 'preview' | 'pinned';
  items: { id: string; label: string; angleDeg: number }[];
  activeIndex: number;
  onHoverIndex?: (i: number) => void;
  onSelectIndex?: (i: number) => void;
  shared: { originX: any; originY: any; scale: any }; // motion values for morph
};

export const KineticDialNav = forwardRef<HTMLDivElement, Props>(function KineticDialNav(
  { id, mode, items, activeIndex, onHoverIndex, onSelectIndex, shared }, ref
) {
  const open = mode !== 'compact';
  const radius = 84; // px; ensure 44px targets with label offset

  return (
    <motion.div
      ref={ref}
      id={id}
      role="navigation"
      aria-hidden={!open}
      className={clsx(
        "relative",
        open ? "pointer-events-auto" : "pointer-events-none"
      )}
      style={{
        originX: shared.originX, // synchronized with Emblem morph
        originY: shared.originY,
        scale: shared.scale
      }}
    >
      {/* Ring */}
      <motion.div
        initial={false}
        animate={{ opacity: open ? 1 : 0, scale: open ? 1 : 0.96 }}
        transition={{ duration: 0.22, ease: [0.33, 1, 0.68, 1] }}     // Orbit feel:contentReference[oaicite:25]{index=25}
        className="relative h-44 w-44"
      >
        {items.map((item, i) => {
          const a = (item.angleDeg * Math.PI) / 180;
          const x = Math.cos(a) * radius;
          const y = Math.sin(a) * radius;
          const isActive = i === activeIndex;

          return (
            <motion.button
              key={item.id}
              className="absolute h-11 min-w-11 grid place-items-center rounded-full"
              style={{ transform: `translate(${x}px, ${y}px)` }}
              tabIndex={open ? (isActive ? 0 : -1) : -1}
              aria-current={isActive ? 'true' : undefined}
              onMouseEnter={() => onHoverIndex?.(i)}
              onFocus={() => onHoverIndex?.(i)}
              onClick={() => onSelectIndex?.(i)}
              // Hit targets ≥ 44px
            >
              <motion.span
                className="px-3 py-2 text-sm text-fg/80 bg-bg/40 backdrop-blur-sm rounded-full"
                initial={false}
                animate={{
                  opacity: open ? 1 : 0,
                  scale: isActive ? 1.02 : 1.0,
                }}
                transition={{ duration: 0.16 }}
              >
                {item.label}
              </motion.span>
            </motion.button>
          );
        })}

        {/* Active arc: emblem progress arc → dial selection arc */}
        <ActiveArc mode={mode} angleDeg={items[activeIndex]?.angleDeg ?? 0} />
      </motion.div>
    </motion.div>
  );
});

function ActiveArc({ mode, angleDeg }: { mode: 'compact'|'preview'|'pinned'; angleDeg: number }) {
  // For MVP: simple rotated wedge. In Phase 2, morph from Emblem’s progress path.
  const open = mode !== 'compact';
  return (
    <motion.div
      className="absolute inset-0"
      style={{ transform: `rotate(${angleDeg - 90}deg)` }} // -90 to start at top
      aria-hidden
    >
      <motion.div
        className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2"
        initial={false}
        animate={{ opacity: open ? 1 : 0 }}
        transition={{ duration: 0.18 }}
      >
        <svg width="176" height="176" viewBox="0 0 176 176">
          <path
            d="M88,8 A80,80 0 0,1 88,168" // half-circle wedge; refine to ±6°
            fill="none"
            stroke="currentColor"
            strokeOpacity="0.6"
            strokeWidth="4"
            strokeLinecap="round"
          />
        </svg>
      </motion.div>
    </motion.div>
  );
}
```
