Here’s the clean way to get a convincing “iris shutter click” in three.js without fighting your SVG paths as literal moving blades: render your logo normally, then animate an iris mask in a shader (polar coordinates + N blades). This also sidesteps your “center is offset” issue because you can center the SVG geometry by its bounding box before masking. (Centering via geometry.center() is a standard fix when rotation/pivots are off.)  ￼

Below are (1) the most relevant references I found, then (2) a copy paste runnable three.js TypeScript module that:
	•	builds an SVG from your EMBLEM_PATHS
	•	loads it via SVGLoader  ￼
	•	recenters it by bounds (fixing your offset)
	•	overlays a shader based iris shutter (N blades, twist, feather)
	•	exposes triggerShutter() for the click animation

// ----- Iris shader overlay -----
function makeIrisOverlayMaterial() {
  const uniforms = {
    uAperture: { value: 1.0 },    // 1 = open, 0 = closed
    uBlades: { value: 10.0 },     // blade count
    uRotation: { value: 0.0 },    // global rotation
    uTwist: { value: 0.65 },      // how much “blade swirl” appears as it closes
    uFeather: { value: 0.02 },    // edge softness
    uOpacity: { value: 1.0 },     // overlay opacity
    uRadiusOpen: { value: 1.25 }, // max opening (in NDC-ish space)
    uRadiusClosed: { value: 0.03 } // min opening
  };

  const vertexShader = /* glsl */ `
    varying vec2 vUv;
    void main() {
      vUv = uv;
      gl_Position = vec4(position.xy, 0.0, 1.0);
    }
  `;

  const fragmentShader = /* glsl */ `
    precision highp float;
    varying vec2 vUv;

    uniform float uAperture;
    uniform float uBlades;
    uniform float uRotation;
    uniform float uTwist;
    uniform float uFeather;
    uniform float uOpacity;
    uniform float uRadiusOpen;
    uniform float uRadiusClosed;

    const float PI = 3.141592653589793;

    float safeAtan(float y, float x) {
      float a = atan(y, x);
      return (a < 0.0) ? (a + 2.0 * PI) : a;
    }

    void main() {
      // map uv to centered coordinates (-1..1)
      vec2 p = vUv * 2.0 - 1.0;

      // polar
      float r = length(p);
      float theta = safeAtan(p.y, p.x);

      // blade math
      float blades = max(3.0, uBlades);
      float w = (2.0 * PI) / blades;

      // add “twist” as it closes to feel like overlapping blades
      float twist = (1.0 - uAperture) * uTwist;
      float t = theta + uRotation + twist;

      // fold angle into one blade sector around 0
      float a = mod(t, w) - 0.5 * w;

      // aperture radius interpolates open->closed
      float baseR = mix(uRadiusClosed, uRadiusOpen, uAperture);

      // regular N-gon boundary: r <= baseR / cos(a)
      // cos(a) can get small; clamp to avoid blowups
      float c = max(0.15, cos(a));
      float limitR = baseR / c;

      // openMask = 1 inside aperture, 0 outside
      float openMask = 1.0 - smoothstep(limitR, limitR + uFeather, r);

      // overlay alpha is inverse of open area (black shutter)
      float alpha = (1.0 - openMask) * uOpacity;

      gl_FragColor = vec4(0.0, 0.0, 0.0, alpha);
    }
  `;

  return new THREE.ShaderMaterial({
    uniforms,
    vertexShader,
    fragmentShader,
    transparent: true,
    depthTest: false,
    depthWrite: false
  });
}

// ----- Build SVG string from your path array -----
function buildSvgString(paths: readonly string[]) {
  const body = paths
    .map((d) => `<path d="${d}" fill="#ffffff" />`)
    .join("");
  return `
    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 588 588">
      ${body}
    </svg>
  `;
}

// ----- Create a centered logo group from the SVG -----
function makeLogoGroupFromPaths(paths: readonly string[]) {
  const svgText = buildSvgString(paths);
  const loader = new SVGLoader();
  const data = loader.parse(svgText);

  const group = new THREE.Group();
  const material = new THREE.MeshBasicMaterial({
    color: 0xffffff,
    side: THREE.DoubleSide,
    depthTest: true
  });

  for (const path of data.paths) {
    // If your SVG paths are fillable, this yields meshes.
    // If a path is not closed, it may produce 0 shapes; that's ok.
    const shapes = SVGLoader.createShapes(path);
    for (const shape of shapes) {
      const geom = new THREE.ShapeGeometry(shape);
      const mesh = new THREE.Mesh(geom, material);
      group.add(mesh);
    }
  }

  // SVG Y axis is flipped vs three.js; flip to match expected orientation
  group.scale.y *= -1;

  // CENTER FIX: center the whole group by its bounds (handles “offset center”)
  const box = new THREE.Box3().setFromObject(group);
  const center = new THREE.Vector3();
  box.getCenter(center);
  group.position.sub(center);

  return group;
}

// ----- Main entry -----
export function mountIrisShutterDemo(opts: { container: HTMLElement }) {
  const { container } = opts;

  const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
  renderer.setPixelRatio(Math.min(2, window.devicePixelRatio));
  container.appendChild(renderer.domElement);

  const scene = new THREE.Scene();

  // Ortho camera with world units that match the shader quad approach
  const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, -10, 10);
  camera.position.z = 5;

  // Logo group (scaled to fit nicely in view)
  const logo = makeLogoGroupFromPaths(EMBLEM_PATHS);
  scene.add(logo);

  // scale logo into the ortho view
  // compute its bounds and scale so the max dimension fits ~1.6 units
  {
    const box = new THREE.Box3().setFromObject(logo);
    const size = new THREE.Vector3();
    box.getSize(size);
    const maxDim = Math.max(size.x, size.y);
    const target = 1.6; // tweak to taste
    const s = maxDim > 0 ? (target / maxDim) : 1;
    logo.scale.multiplyScalar(s);
  }

  // Iris overlay plane (full screen quad in clip space)
  const irisMat = makeIrisOverlayMaterial();
  const irisPlane = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), irisMat);
  irisPlane.position.z = 2;
  scene.add(irisPlane);

  // Resize
  function resize() {
    const w = container.clientWidth;
    const h = container.clientHeight;
    renderer.setSize(w, h, false);
  }
  window.addEventListener("resize", resize);
  resize();

  // Click animation state
  let shutterT = 1.0;
  let shutterActive = false;
  let shutterDir: "closing" | "opening" = "closing";

  function triggerShutter() {
    // start from open
    shutterT = 1.0;
    shutterActive = true;
    shutterDir = "closing";
  }

  // Example trigger
  container.addEventListener("pointerdown", () => triggerShutter());

  const clock = new THREE.Clock();

  function tick() {
    const dt = Math.min(0.05, clock.getDelta());
    const t = clock.elapsedTime;

    // subtle rotation makes blades feel mechanical
    (irisMat.uniforms.uRotation.value as number) = t * 0.6;

    // shutter “click”: fast close, slightly slower reopen
    if (shutterActive) {
      if (shutterDir === "closing") {
        shutterT -= dt * 6.5; // close speed
        if (shutterT <= 0.0) {
          shutterT = 0.0;
          shutterDir = "opening";
        }
      } else {
        shutterT += dt * 4.0; // open speed
        if (shutterT >= 1.0) {
          shutterT = 1.0;
          shutterActive = false;
        }
      }

      // ease curve so it “snaps”
      const eased = shutterT * shutterT * (3.0 - 2.0 * shutterT);
      (irisMat.uniforms.uAperture.value as number) = eased;
    } else {
      (irisMat.uniforms.uAperture.value as number) = 1.0;
    }

    renderer.render(scene, camera);
    requestAnimationFrame(tick);
  }

  tick();

  return {
    renderer,
    scene,
    camera,
    logo,
    triggerShutter,
    setBladeCount(n: number) {
      irisMat.uniforms.uBlades.value = Math.max(3, Math.floor(n));
    },
    setFeather(f: number) {
      irisMat.uniforms.uFeather.value = Math.max(0.0, f);
    }
  };
}

Why your “center is offset” (and why this fixes it)

SVG path coordinates live in the SVG’s coordinate space, and your combined emblem’s visual center is not guaranteed to be the viewBox midpoint. If you rotate around (0,0) without centering the geometry/group, it will orbit instead of spin in place. Centering by bounding box (or geometry.center() for a single geometry) is the standard correction.  ￼
