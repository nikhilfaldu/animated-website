	•	Toggle pointer events + opacity + render loop.
	•	In R3F: set frameloop="demand" and call invalidate() only during 65–100% (or while in view). That gives you the perf benefit without the startup stutter.

2) The “offscreen radius ~800px” is the one part that can look generic

If the rays come from far away, it starts reading like a standard assembly move.

	•	Keep rays roughly in place radially, but rotate them “closed” (like shutter blades) and slightly scale in.
	•	Add a micro tangential slip during the first 20% (each ray drifts a few pixels along the circumference, then snaps back). That makes it feel like indexing teeth finding their seat.

3) Stagger: avoid “domino” sequencing

A uniform 12ms per ray can look like a typical cascade.

emblemTexture.ts

import * as THREE from 'three';
import { EMBLEM_PATHS } from '@/assets/emblemPaths';

/**
 * Builds a CanvasTexture by rendering your emblem SVG paths.
 * This keeps the WebGL emblem pixel-faithful to your actual logo.
 */
export async function createEmblemTexture(size = 2048): Promise<THREE.CanvasTexture> {
  const svg = `
  <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 588 588" width="${size}" height="${size}">
    <rect width="100%" height="100%" fill="transparent"/>
    <g fill="white">
      ${EMBLEM_PATHS.map((d) => `<path d="${d}" />`).join('')}
    </g>
  </svg>`.trim();

  const img = new Image();
  img.decoding = 'async';
  img.src = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`;

  await new Promise<void>((resolve, reject) => {
    img.onload = () => resolve();
    img.onerror = () => reject(new Error('Failed to load emblem SVG into Image()'));
  });

  const canvas = document.createElement('canvas');
  canvas.width = size;
  canvas.height = size;

  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('No 2D canvas context available');

  ctx.clearRect(0, 0, size, size);
  ctx.drawImage(img, 0, 0, size, size);

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = THREE.ClampToEdgeWrapping;
  tex.wrapT = THREE.ClampToEdgeWrapping;
  tex.minFilter = THREE.LinearMipmapLinearFilter;
  tex.magFilter = THREE.LinearFilter;
  tex.generateMipmaps = true;
  tex.needsUpdate = true;

  return tex;
}

apertureShader.ts

export const vertexShader = `
  varying vec2 vUv;
  void main() {
    vUv = uv;
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
  }
`;

export const fragmentShader = `
  precision highp float;

  uniform sampler2D uTex;
  uniform float uProgress;   // 0..1
  uniform float uOpacity;    // 0..1
  uniform float uSwingMax;   // radians (e.g. 1.57 for 90deg)
  uniform float uRadialPush; // how far “out” the rays start
  uniform float uHinge;      // hinge radius in normalized coords (0..1)
  uniform float uDelays[32]; // per-slice delay 0..~0.18

  varying vec2 vUv;

  const float PI = 3.141592653589793;
  const float TAU = 6.283185307179586;
  const float SLICES = 32.0;

  mat2 rot(float a) {
    float s = sin(a);
    float c = cos(a);
    return mat2(c, -s, s, c);
  }

  float saturate(float x) { return clamp(x, 0.0, 1.0); }

  void main() {
    // centered coords in [-1,1]
    vec2 c = (vUv - 0.5) * 2.0;

    // polar angle in [0, 2pi)
    float theta = atan(c.y, c.x);
    if (theta < 0.0) theta += TAU;

    float sliceW = TAU / SLICES;
    float slice = floor(theta / sliceW);
    float sliceCenter = (slice + 0.5) * sliceW;

    int idx = int(slice);
    float d = uDelays[idx];

    // local progress per slice (staggered), then ease
    float p = saturate((uProgress - d) / max(0.0001, (1.0 - d)));
    p = smoothstep(0.0, 1.0, p);

    // Move into slice frame (slice center aligned to +X)
    vec2 cs = rot(-sliceCenter) * c;

    // Hinge point along +X axis near center
    vec2 hinge = vec2(uHinge, 0.0);

    // Start “closed/out” and swing into place
    float swing = (1.0 - p) * uSwingMax;

    // Radial push outward at start to amplify the “return”
    cs.x += uRadialPush * (1.0 - p);

    // Rotate around hinge
    vec2 rel = cs - hinge;
    rel = rot(swing) * rel;
    cs = rel + hinge;

    // Back to world frame
    vec2 cw = rot(sliceCenter) * cs;

    // Back to UV
    vec2 uv2 = (cw * 0.5) + 0.5;

    // Out of bounds -> transparent
    if (uv2.x < 0.0 || uv2.x > 1.0 || uv2.y < 0.0 || uv2.y > 1.0) {
      gl_FragColor = vec4(0.0);
      return;
    }

    vec4 tex = texture2D(uTex, uv2);

    // crisp threshold (keeps the rays clean)
    float a = smoothstep(0.15, 0.35, tex.a);

    gl_FragColor = vec4(vec3(1.0), a * uOpacity);
  }
`;

useApertureUniforms.ts

import { useEffect, useMemo, useRef } from 'react';
import * as THREE from 'three';
import { useMotionValueEvent, useSpring, useTransform } from 'framer-motion';
import { APERTURE_START, APERTURE_END } from '@/sections/Process/processData';
import { createEmblemTexture } from './emblemTexture';

function clamp01(n: number) {
  return Math.max(0, Math.min(1, n));
}

export function useApertureUniforms(scrollProgress: any) {
  const uniforms = useMemo(() => {
    const delays = new Float32Array(32);

    // Quadrant-distributed stagger with slight deterministic jitter.
    // This avoids a visible wipe without relying on Math.random().
    const order: number[] = [];
    const quadrants = [
      Array.from({ length: 8 }, (_, i) => i),
      Array.from({ length: 8 }, (_, i) => i + 8),
      Array.from({ length: 8 }, (_, i) => i + 16),
      Array.from({ length: 8 }, (_, i) => i + 24),
    ];

    // Deterministic shuffle-ish: interleave with a stride
    const stride = 3;
    const shuffled = quadrants.map((q) => q.map((_, i) => q[(i * stride) % q.length]));

    for (let i = 0; i < 8; i++) {
      order.push(shuffled[0][i], shuffled[2][i], shuffled[1][i], shuffled[3][i]);
    }

    order.forEach((rayIdx, i) => {
      const base = i * 0.010; // 10ms base spacing
      const jitter = (rayIdx % 7) * 0.002; // 0..12ms deterministic jitter
      delays[rayIdx] = Math.min(0.22, base + jitter);
    });

    return {
      uTex: { value: null as unknown as THREE.Texture },
      uProgress: { value: 0 },
      uOpacity: { value: 0 },
      uSwingMax: { value: Math.PI / 2 }, // 90°
      uRadialPush: { value: 0.55 },      // normalized push
      uHinge: { value: 0.08 },           // hinge close to center
      uDelays: { value: Array.from(delays) },
    };
  }, []);

  // Map local section scroll -> 0..1 aperture progress
  const raw = useTransform(scrollProgress, (v: number) => {
    const p = (v - APERTURE_START) / (APERTURE_END - APERTURE_START);
    return clamp01(p);
  });

  // Spring it so the “index lock” feels physical
  const spring = useSpring(raw, { stiffness: 180, damping: 22, mass: 1 });

  // Drive opacity: invisible before start, fully visible during main action
  const opacityMv = useTransform(scrollProgress, (v: number) => {
    if (v < APERTURE_START - 0.02) return 0;
    if (v < APERTURE_START) return (v - (APERTURE_START - 0.02)) / 0.02;
    if (v > APERTURE_END + 0.03) return 0;
    return 1;
  });

  // Load texture once
  const textureLoadedRef = useRef(false);
  useEffect(() => {
    if (textureLoadedRef.current) return;
    textureLoadedRef.current = true;

    (async () => {
      const tex = await createEmblemTexture(2048);
      uniforms.uTex.value = tex;
    })().catch((err) => {
      console.error(err);
    });
  }, [uniforms]);

  // Pipe MotionValues into uniforms (no per-frame React state)
  useMotionValueEvent(spring, 'change', (v) => {
    uniforms.uProgress.value = v;
  });

  useMotionValueEvent(opacityMv, 'change', (v) => {
    uniforms.uOpacity.value = v;
  });

  return uniforms;
}

AperturePlane.tsx

import * as THREE from 'three';
import { useEffect, useMemo } from 'react';
import { useFrame, useThree } from '@react-three/fiber';
import { fragmentShader, vertexShader } from './apertureShader';
import { useApertureUniforms } from './useApertureUniforms';

type Props = {
  scrollProgress: any;
};

export function AperturePlane({ scrollProgress }: Props) {
  const { invalidate, viewport } = useThree();
  const uniforms = useApertureUniforms(scrollProgress);

  const material = useMemo(() => {
    return new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      depthTest: false,
      uniforms,
      vertexShader,
      fragmentShader,
    });
  }, [uniforms]);

  // Fullscreen plane in orthographic space
  // We size it to cover the current viewport.
  const planeArgs = useMemo<[number, number]>(() => {
    return [viewport.width, viewport.height];
  }, [viewport.width, viewport.height]);

  // Demand-driven rendering: invalidate when opacity > 0
  useEffect(() => {
    const id = setInterval(() => {
      if (uniforms.uOpacity.value > 0.001) invalidate();
    }, 16);
    return () => clearInterval(id);
  }, [invalidate, uniforms]);

  // Keep a tiny render heartbeat while visible to stay buttery during scroll
  useFrame(() => {
    if (uniforms.uOpacity.value > 0.001) invalidate();
  });

  return (
    <mesh material={material}>
      <planeGeometry args={planeArgs} />
    </mesh>
  );
}
